import java.util.Scanner;

class compute_palindrome_string implements Runnable {
    private final String value;
    private boolean isPalindrome;

    compute_palindrome_string(String value) {
        // TODO 1: assign constructor parameter to the field
        this.value = value;
    }

    // Case-sensitive palindrome check (no normalization)
    private boolean isPalindrome(String a) {

        int len = /* TODO 2: length of a */ a.length();

        if (len == 0) {
            System.out.println("Empty String");
            return false;
        }
        if (len == 1) {
            System.out.println("Its a palindrome");
            return true;
        }

        for (int i = 0; i < /* TODO 3: loop half length */ len / 2; i++) {
            // TODO 4: compare symmetric characters; if not equal -> print and return false
            if (a.charAt(i) != a.charAt(len - 1 - i)) {
                System.out.println("Not a palindrome");
                return false;
            }
        }
        System.out.println("Its a palindrome");
        return true;
    }

    @Override
    public void run() {
        // TODO 5: call checker and assign to isPalindrome
        isPalindrome = isPalindrome(value);
    }

    boolean getResult() {
        return isPalindrome;
    }
}

public class palindrome {
    public static void main(String[] args) throws InterruptedException {
        try (Scanner sc = new Scanner(System.in)) {

            // TODO 6: create the task
            compute_palindrome_string task = new compute_palindrome_string(line);

            // TODO 7: wrap Runnable into a Thread
            Thread t = new Thread(task);

            // TODO 8: start the thread
            t.start();

            // TODO 9: wait for completion
            t.join();

            boolean result = task.getResult();
            // TODO 10: print the final verdict exactly as specified
            System.out.println("\"" + line + "\"" + (result ? " is" : " is not") + " a palindrome.");
        }
    }
}
